        // ================== CONFIG ==================
        let soundEnabled = true;

        let gameOverCount = 0;

        const COLORS = [
            {name: "RED",    hex: "#FF4D6E"},
            {name: "BLUE",   hex: "#4D9CFF"},
            {name: "GREEN",  hex: "#00E68C"},
            {name: "YELLOW", hex: "#FFEA4D"},

        ];

        let score = 0, combo = 1, bestScore = 0;
        let timeLeft = 4, baseTime = 4000;
        let timerInterval = null;
        let currentWord = "", currentColorHex = "";
        let correctCount = 0, totalAnswered = 0;
        let gameState = "menu";

        let particles = [];
        let canvas, ctx;
        let audioCtx;

        const PAYPAL_LINK = "https://paypal.me/Solacior";

        let achievements = {
            firstScore: { name: "First Steps", desc: "Score 100 points", unlocked: false },
            score500:   { name: "Rising Star", desc: "Reach 500 points", unlocked: false },
            score1500:  { name: "Color Master", desc: "Reach 1500 points", unlocked: false },
            combo8:     { name: "Combo Expert", desc: "Reach 8x combo", unlocked: false },
            perfect10:  { name: "Flawless", desc: "10 correct in a row", unlocked: false },
            dailyWin:   { name: "Daily Champion", desc: "Complete Daily Challenge", unlocked: false }
        };

        let daily = { date: "", target: 1200, current: 0, completed: false };

        function playSound(type) {
            if (!soundEnabled) return;
            if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
            const osc = audioCtx.createOscillator();
            const gain = audioCtx.createGain();
            osc.connect(gain).connect(audioCtx.destination);
            if (type === "correct") { osc.frequency.value = 920; gain.gain.value = 0.3; }
            else if (type === "wrong") { osc.frequency.value = 160; gain.gain.value = 0.45; osc.type = "sawtooth"; }
            osc.start();
            osc.stop(audioCtx.currentTime + 0.45);
        }

        class Particle {
            constructor(x, y, vx, vy, color, life, size, confetti = false) {
                this.x = x; this.y = y; this.vx = vx; this.vy = vy;
                this.color = color; this.life = life; this.maxLife = life;
                this.size = size; this.confetti = confetti;
                this.angle = Math.random() * Math.PI * 2;
                this.spin = confetti ? (Math.random() - 0.5) * 0.4 : 0;
            }
            update() {
                this.vy += 0.16;
                this.x += this.vx;
                this.y += this.vy;
                this.life--;
                this.angle += this.spin;
            }
            draw() {
                const a = this.life / this.maxLife;
                ctx.save();
                ctx.globalAlpha = a;
                ctx.translate(this.x, this.y);
                ctx.rotate(this.angle);
                ctx.fillStyle = this.color;
                if (this.confetti) ctx.fillRect(-this.size*2, -this.size*0.7, this.size*4, this.size*1.4);
                else {
                    ctx.shadowBlur = 15;
                    ctx.shadowColor = this.color;
                    ctx.beginPath(); ctx.arc(0,0,this.size,0,Math.PI*2); ctx.fill();
                }
                ctx.restore();
            }
        }

        function burst(x, y) {
            for (let i = 0; i < 75; i++) {
                const angle = Math.random() * Math.PI * 2;
                const vel = 4 + Math.random() * 11;
                const col = COLORS[Math.floor(Math.random()*COLORS.length)].hex;
                particles.push(new Particle(x, y, Math.cos(angle)*vel, Math.sin(angle)*vel - 5, col, 48, 6));
            }
        }

        function loadProgress() {
            bestScore = parseInt(localStorage.getItem("forbiddenBest") || "0");
            document.getElementById("best").textContent = bestScore;

            const savedAch = localStorage.getItem("forbiddenAchievements");
            if (savedAch) achievements = JSON.parse(savedAch);

            const today = new Date().toISOString().slice(0,10);
            const savedDaily = localStorage.getItem("forbiddenDaily");
            if (savedDaily) {
                const d = JSON.parse(savedDaily);
                if (d.date === today) daily = d;
                else resetDaily(today);
            } else resetDaily(today);
        }

        function resetDaily(today) {
            daily = { date: today, target: 1000 + Math.floor(Math.random()*800), current: 0, completed: false };
            localStorage.setItem("forbiddenDaily", JSON.stringify(daily));
        }

        function unlockAchievement(key) {
            if (achievements[key].unlocked) return;
            achievements[key].unlocked = true;
            localStorage.setItem("forbiddenAchievements", JSON.stringify(achievements));

            const popup = document.createElement("div");
            popup.style.cssText = `position:fixed;bottom:120px;left:50%;transform:translateX(-50%);background:linear-gradient(90deg,#ffd4a3,#ff9ed8);color:#111;padding:16px 32px;border-radius:9999px;font-weight:900;box-shadow:0 15px 40px rgba(0,0,0,0.4);z-index:200;`;
            popup.textContent = `🏆 ${achievements[key].name} UNLOCKED!`;
            document.body.appendChild(popup);
            setTimeout(() => popup.remove(), 2800);
        }

        function checkAchievements() {
            if (score >= 100 && !achievements.firstScore.unlocked) unlockAchievement("firstScore");
            if (score >= 500 && !achievements.score500.unlocked) unlockAchievement("score500");
            if (score >= 1500 && !achievements.score1500.unlocked) unlockAchievement("score1500");
            if (combo >= 8 && !achievements.combo8.unlocked) unlockAchievement("combo8");
            if (correctCount >= 10 && correctCount === totalAnswered && !achievements.perfect10.unlocked) unlockAchievement("perfect10");
        }

        function createColorButtons() {
            const grid = document.getElementById("color-grid");
            grid.innerHTML = "";
            COLORS.forEach(c => {
                const btn = document.createElement("button");
                btn.className = `color-btn ${c.name.toLowerCase()}`;
                btn.style.backgroundColor = c.hex;
                btn.textContent = c.name;
                btn.addEventListener("click", () => handleAnswer(c.hex));
                grid.appendChild(btn);
            });
        }

        function pickNewRound() {
            const wordIdx = Math.floor(Math.random() * COLORS.length);
            currentWord = COLORS[wordIdx].name;
            let colorIdx = Math.floor(Math.random() * COLORS.length);
            if (colorIdx === wordIdx && Math.random() < 0.78) colorIdx = (colorIdx + 3) % COLORS.length;

            currentColorHex = COLORS[colorIdx].hex;
            const wordEl = document.getElementById("word");
            wordEl.textContent = currentWord;
            wordEl.style.color = currentColorHex;
        }

        function startTimer() {
            if (timerInterval) clearInterval(timerInterval);
            timeLeft = baseTime / 1000;
            document.getElementById("timer").textContent = timeLeft.toFixed(1);

            timerInterval = setInterval(() => {
                timeLeft -= 0.1;
                document.getElementById("timer").textContent = Math.max(0, timeLeft).toFixed(1);
                if (timeLeft <= 0) endGame();
            }, 100);
        }

        function handleAnswer(selectedHex) {
            if (gameState !== "playing") return;
            totalAnswered++;

            if (selectedHex === currentColorHex) {
                correctCount++;
                const points = 100 * combo;
                score += points;
                playSound("correct");

                const wordEl = document.getElementById("word");
                wordEl.style.transform = "scale(1.28)";
                setTimeout(() => wordEl.style.transform = "scale(1)", 180);

                const rect = wordEl.getBoundingClientRect();
                burst(rect.left + rect.width/2, rect.top + rect.height/2);

                combo++;
                document.getElementById("score").textContent = score;

                baseTime = Math.max(1600, baseTime - 110);
                clearInterval(timerInterval);
                pickNewRound();
                startTimer();
                checkAchievements();
            } else {
                playSound("wrong");
                endGame();
            }
        }

        function endGame() {
            gameState = "over";
            clearInterval(timerInterval);
            if (score > bestScore) {
                bestScore = score;
                localStorage.setItem("forbiddenBest", bestScore);
                document.getElementById("best").textContent = bestScore;
            }

            document.getElementById("final-score").textContent = score;
            const accuracy = totalAnswered ? Math.round((correctCount / totalAnswered) * 100) : 0;
            document.getElementById("accuracy").innerHTML = `Accuracy: <strong>${accuracy}%</strong> • Combo: ×${combo-1}`;

            gameOverCount++;
            document.getElementById("game-over-modal").style.display = "flex";

            if (gameOverCount % 3 === 0) {
                setTimeout(() => { try { adBreak({ type: 'next', name: 'after_game_over' }); } catch(e){} }, 800);
            }
        }

        function watchAdForContinue() {
            const btn = document.querySelector("#game-over-modal button[onclick='watchAdForContinue()']");
            btn.textContent = "Loading Ad...";
            try {
                adBreak({
                    type: 'reward',
                    name: 'continue_after_game_over',
                    adViewed: () => giveExtraLife(),
                    adDismissed: () => { btn.textContent = "🎥 WATCH AD FOR EXTRA LIFE (+3s)"; },
                    adBreakDone: (info) => { if (info.breakStatus === "viewed") giveExtraLife(); }
                });
            } catch(e) { giveExtraLife(); }
        }

        function giveExtraLife() {
            document.getElementById("game-over-modal").style.display = "none";
            baseTime += 3000;
            gameState = "playing";
            pickNewRound();
            startTimer();
            burst(canvas.width/2, canvas.height/2);
        }

        function startGame(isDaily = false) {
            document.getElementById("menu-screen").style.display = "none";
            document.getElementById("game-screen").style.display = "block";

            score = 0; combo = 1; baseTime = isDaily ? 3800 : 4000;
            correctCount = 0; totalAnswered = 0;
            document.getElementById("score").textContent = "0";

            gameState = "playing";
            createColorButtons();
            pickNewRound();
            startTimer();
        }

        function startDailyChallenge() {
            closeModal('daily-modal');
            startGame(true);
        }

        function showDailyChallenge() {
            const today = new Date().toISOString().slice(0,10);
            if (daily.date !== today) resetDaily(today);

            document.getElementById("daily-goal").innerHTML = `Reach <strong>${daily.target}</strong> points today!`;
            const percent = Math.min(100, Math.round((daily.current / daily.target) * 100));
            document.getElementById("daily-bar").style.width = percent + "%";
            document.getElementById("daily-modal").style.display = "flex";
        }

        function showAchievements() {

            list.innerHTML = "";
            Object.keys(achievements).forEach(key => {
                const ach = achievements[key];

                div.innerHTML = `<strong>${ach.name}</strong><br><small>${ach.desc}</small>`;
                if (ach.unlocked) div.innerHTML += ' <span style="color:#0f0;">✓ UNLOCKED</span>';
                list.appendChild(div);
            });
            document.getElementById("achievements-modal").style.display = "flex";
        }

        function showThankYou() {
            const thanks = document.createElement("div");
            thanks.style.cssText = `position:fixed;bottom:100px;left:50%;transform:translateX(-50%);background:#ffd4a3;color:#111;padding:16px 32px;border-radius:9999px;font-weight:900;box-shadow:0 15px 40px rgba(0,0,0,0.4);z-index:200;`;

        function closeModal(id) {
            document.getElementById(id).style.display = "none";
        }

        function toggleSound() {
            soundEnabled = !soundEnabled;
            const btn = document.getElementById("sound-btn");
            const pauseBtn = document.getElementById("pause-sound-btn");
            if (btn) btn.textContent = soundEnabled ? "🔊" : "🔇";
            if (pauseBtn) pauseBtn.textContent = "Sound: " + (soundEnabled ? "ON" : "OFF");
            localStorage.setItem("forbiddenSound", soundEnabled);
        }

        function toggleTheme() {
            currentTheme = currentTheme === "light" ? "dark" : "light";
            document.body.setAttribute("data-theme", currentTheme);
            localStorage.setItem("forbiddenTheme", currentTheme);
        }

        function pauseGame() {
            if (gameState !== "playing") return;
            gameState = "paused";
            clearInterval(timerInterval);
            document.getElementById("pause-modal").style.display = "flex";
        }

        function resumeGame() {
            document.getElementById("pause-modal").style.display = "none";
            gameState = "playing";
            startTimer();
        }

        function quitToMenu() {
            document.getElementById("game-over-modal").style.display = "none";
            document.getElementById("pause-modal").style.display = "none";
            document.getElementById("game-screen").style.display = "none";
            document.getElementById("menu-screen").style.display = "flex";
            gameState = "menu";
        }

        function restartGame() {
            document.getElementById("game-over-modal").style.display = "none";
            startGame();
        }

        function init() {
            canvas = document.getElementById("particles");
            ctx = canvas.getContext("2d");

            const resize = () => { canvas.width = window.innerWidth; canvas.height = window.innerHeight; };
            resize();
            window.addEventListener("resize", resize);

            loadProgress();

            if (localStorage.getItem("forbiddenSound") !== null) soundEnabled = localStorage.getItem("forbiddenSound") === "true";
            if (localStorage.getItem("forbiddenTheme")) {
                currentTheme = localStorage.getItem("forbiddenTheme");
                document.body.setAttribute("data-theme", currentTheme);
            }

            document.getElementById("sound-btn").textContent = soundEnabled ? "🔊" : "🔇";

            function animate() {
                ctx.clearRect(0, 0, canvas.width, canvas.height);
                for (let i = particles.length - 1; i >= 0; i--) {
                    particles[i].update();
                    particles[i].draw();
                    if (particles[i].life <= 0) particles.splice(i, 1);
                }
                requestAnimationFrame(animate);
            }
            animate();
        }

        window.onload = init;
    </script>
